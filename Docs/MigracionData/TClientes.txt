CREATE INDEX "CLT_KEYCOBRADOR"
  ON "TClientes"
  ("CLT_COBRADOR", "CLT_STATUS");

CREATE INDEX "CLT_KEYDESCRIPCION"
  ON "TClientes"
  ("CLT_DESCRIPCION");

CREATE INDEX "CLT_KEYMONEDA"
  ON "TClientes"
  ("CLT_MONEDA", "CLT_CODIGO");

CREATE INDEX "CLT_KEYTIPOCLT"
  ON "TClientes"

  ON "TClientes"
  ("CLT_VENDEDOR", "CLT_STATUS");
